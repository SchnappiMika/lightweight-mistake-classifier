# ===========================================================================
# 大学生轻量化错题自动分类管理系统 —— 依赖清单
#
#   pip install -r requirements.txt
#
# 核心文本分类本地运行；可选识别引擎首次初始化可能下载模型，离线能力需另行验证。
# 目标 Python 版本：3.9+；当前 Windows 验证环境为 3.13.5，3.9 未实测
# ===========================================================================

# --- 核心依赖（必需，缺一不可） ---------------------------------------------
jieba>=0.42.1            # 中文分词，支持自定义专有名词词典
scikit-learn>=1.0        # TF-IDF 词表构建与余弦相似度
numpy>=1.20              # 向量运算
scipy>=1.7               # 稀疏矩阵（标签画像矩阵）
pandas>=1.3              # Excel / CSV 读写
openpyxl>=3.0            # pandas 的 xlsx 引擎
PyQt5>=5.15              # 桌面界面

# --- 可选依赖（不装也能用，只是对应模态降级） -------------------------------
#
# 这三个都是体积较大的识别引擎。系统的设计前提就是"没装也能跑"：
#   * 题干里用 $...$ 手写的公式    -> 任何时候都能正常识别与分类
#   * 图形题 / 听力题的人工标注    -> 任何时候都能正常参与分类
# 装上之后才会多出"从图片认公式""从图里认文字""从音频转文字"的能力。
# 程序内「环境自检」只查询组件安装信息与已有加载状态，不主动加载模型或下载。
#
# 按需取用；安装组件后还需准备模型并用真实文件验证，自检不会替代识别验收：
#
# 公式图片 -> LaTeX（约 1.5GB，含 PyTorch）
#   pip install "pix2tex[gui]"
#
# 印刷中文/英文图片 -> 文字校对（当前默认方案，支持本机 Python 3.13）
#   双击 安装图片识别.bat，或在项目 .venv 内 pip install -r requirements-ocr.txt
#   RapidOCR wheel 内含约 30.3 MiB 本地模型；识别前校验，不隐式下载。
#
# 旧图形文字适配器（非当前默认，不需要同时安装，未完成独立验收）
#   pip install paddlepaddle paddleocr
#
# 英语听力音频 -> 文字（faster-whisper 比官方 whisper 更省内存，推荐）
#   pip install faster-whisper
#   # 或：pip install openai-whisper
#
# 英文词形还原（不装则自动退化为规则版后缀剥离，准确率略降）
#   pip install nltk
#   python -c "import nltk; nltk.download('wordnet'); nltk.download('punkt')"

# 听力转写的可选依赖已单独锁定在 requirements-media.txt，避免核心安装拉取大组件。
